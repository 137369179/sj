# 市集招募平台 MVP 招募闭环 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 构建市集招募平台的首个可上线 MVP，跑通“主办方发市集 -> 摊主浏览 -> 在线报名 -> 主办方审核 -> 摊位分配 -> 通知结果 -> 基础看板”的完整闭环。

**Architecture:** 采用 `Next.js App Router + TypeScript` 单仓实现，先交付 `主办方 PC 后台 + 摊主 H5/PWA + 平台最小审核端`。后端采用模块化单体，按 `auth / markets / applications / stalls / notifications / dashboard / admin-review` 分域，数据层使用 `Prisma + PostgreSQL`，并为对象存储和 Redis 留出适配点。

**Tech Stack:** Next.js, TypeScript, Tailwind CSS, Prisma, PostgreSQL, Zod, Vitest, Testing Library, Playwright, pnpm

```ts
// apps/web/src/lib/roles.ts
export type UserRole = "vendor" | "organizer" | "admin";

const routeRules: Array<{ prefix: string; roles: UserRole[] }> = [
  { prefix: "/organizer", roles: ["organizer", "admin"] },
  { prefix: "/admin", roles: ["admin"] },
  { prefix: "/markets", roles: ["vendor", "organizer", "admin"] }
];

export function canAccessRoute(role: UserRole, pathname: string) {
  const matched = routeRules.find((rule) => pathname.startsWith(rule.prefix));
  if (!matched) return true;
  return matched.roles.includes(role);
}
```

```ts
// apps/web/src/lib/auth.ts
import { cookies } from "next/headers";
import type { UserRole } from "./roles";

const SESSION_KEY = "mrp_session_role";

export async function getSessionRole(): Promise<UserRole | null> {
  const cookieStore = await cookies();
  const role = cookieStore.get(SESSION_KEY)?.value;
  return role === "vendor" || role === "organizer" || role === "admin" ? role : null;
}
```

```ts
// apps/web/src/app/api/auth/login/route.ts
import { NextResponse } from "next/server";

export async function POST(request: Request) {
  const body = await request.json();
  const role = body.role;

  if (!["vendor", "organizer", "admin"].includes(role)) {
    return NextResponse.json({ message: "invalid role" }, { status: 400 });
  }

  const response = NextResponse.json({ ok: true });
  response.cookies.set("mrp_session_role", role, { httpOnly: true, path: "/" });
  return response;
}
```

```ts
// apps/web/src/middleware.ts
import { NextResponse } from "next/server";
import type { NextRequest } from "next/server";
import { canAccessRoute } from "./lib/roles";

export function middleware(request: NextRequest) {
  const role = request.cookies.get("mrp_session_role")?.value as
    | "vendor"
    | "organizer"
    | "admin"
    | undefined;

  if (role && !canAccessRoute(role, request.nextUrl.pathname)) {
    return NextResponse.redirect(new URL("/", request.url));
  }

  return NextResponse.next();
}

export const config = {
  matcher: ["/organizer/:path*", "/admin/:path*", "/markets/:path*"]
};
```

```ts
import { NextResponse } from "next/server";
import { db } from "@/lib/db";
import { canPublishMarket } from "@/server/markets/service";

export async function POST(
  _request: Request,
  { params }: { params: Promise<{ marketId: string }> }
) {
  const { marketId } = await params;
  const market = await db.market.findUnique({ where: { id: marketId } });

  if (!market || !canPublishMarket(market.status)) {
    return NextResponse.json({ message: "cannot publish" }, { status: 400 });
  }

  const updated = await db.market.update({
    where: { id: marketId },
    data: { status: "published" }
  });

  return NextResponse.json(updated);
}
```

```ts
// apps/web/src/server/markets/service.ts
export function filterMarkets<T extends { city: string; title: string }>(
  markets: T[],
  filters: { city?: string; keyword?: string }
) {
  return markets.filter((market) => {
    const cityMatched = filters.city ? market.city === filters.city : true;
    const keywordMatched = filters.keyword
      ? market.title.includes(filters.keyword)
      : true;
    return cityMatched && keywordMatched;
  });
}
```

```ts
// apps/web/src/server/applications/service.ts
export function reviewApplication(
  application: { id: string; status: string },
  input: { decision: "approved" | "rejected"; note?: string }
) {
  if (application.status !== "under_review") {
```

```ts
// apps/web/src/app/api/applications/[applicationId]/review/route.ts
import { NextResponse } from "next/server";
import { db } from "@/lib/db";
import { reviewApplication } from "@/server/applications/service";
import { buildReviewNotification } from "@/server/notifications/service";

export async function POST(
  request: Request,
  { params }: { params: Promise<{ applicationId: string }> }
) {
  const { applicationId } = await params;
  const body = await request.json();
  const current = await db.application.findUnique({ where: { id: applicationId } });

  if (!current) {
    return NextResponse.json({ message: "not found" }, { status: 404 });
  }

  const reviewed = reviewApplication(current, body);
  const updated = await db.application.update({
    where: { id: applicationId },
    data: {
      status: reviewed.status,
      note: reviewed.note
    }
  });

  await db.notification.create({
    data: {
      userId: current.vendorId,
      ...buildReviewNotification(body.decision)
    }
  });

  return NextResponse.json(updated);
}
```

```ts
  request: Request,
  { params }: { params: Promise<{ stallId: string }> }
) {
  const { stallId } = await params;
  const body = await request.json();
  const stall = await db.stall.findUnique({ where: { id: stallId } });
```
